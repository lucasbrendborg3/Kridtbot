
if(CMAKE_CXX_COMPILER_ID MATCHES "(GNU|Clang)")
  add_compile_options(-Wall -Wextra)
endif()

# Find dependencies
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(std_srvs REQUIRED)

include_directories(
  include
  ${rclcpp_INCLUDE_DIRS}
  ${sensor_msgs_INCLUDE_DIRS}
  ${std_msgs_INCLUDE_DIRS}
  ${diagnostic_msgs_INCLUDE_DIRS}
)

add_library(imu_bno055 SHARED
    src/bno055_i2c_driver.cpp  # Include the source file with the constructor definition
)

add_executable(bno055_i2c_node
  src/bno055_i2c_driver.cpp
  src/bno055_i2c_node_ros2.cpp
  src/watchdog.cpp
)

ament_target_dependencies(bno055_i2c_node
  rclcpp
  sensor_msgs
  std_msgs
  diagnostic_msgs
  std_srvs
)

install(TARGETS bno055_i2c_node
  DESTINATION lib/${PROJECT_NAME}
)

install(
  DIRECTORY include/
  DESTINATION include
)

install(TARGETS imu_bno055
  EXPORT imu_bno055_export
  DESTINATION lib
)

install(EXPORT imu_bno055_export
  FILE imu_bno055Config.cmake
  DESTINATION share/ament_index/resource_index/ament_cmake/
)


ament_export_include_directories(include)
ament_export_libraries(imu_bno055)

ament_package()
